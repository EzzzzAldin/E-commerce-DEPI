STEP 8. Showing Discount Inside ProductCard
function ProductCard({ discount = 0 }) {
  return (
    <div className="price flex items-center justify-between">
      {discount > 0 && (
        <span className="sale py-0.75 px-1.5 rounded-xs bg-red-off text-[8px] font-semibold text-white">
          -{discount}%
        </span>
      )}
    </div>
  );
}

STEP 9. Separating Discount Logic into DiscountBadge
function DiscountBadge({ discount }) {
  if (discount <= 0) {
    return null;
  }

  return (
    <span className="sale py-0.75 px-1.5 rounded-xs bg-red-off text-[8px] font-semibold text-white">
      -{discount}%
    </span>
  );
}
